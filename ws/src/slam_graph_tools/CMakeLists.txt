cmake_minimum_required(VERSION 3.16)
project(slam_graph_tools)
find_package(ament_cmake REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization)
find_package(TBB REQUIRED)
find_library(KARTO_LIBRARY kartoSlamToolbox PATHS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
add_executable(export_slam_graph src/export_slam_graph.cpp)
target_compile_features(export_slam_graph PUBLIC cxx_std_17)
ament_target_dependencies(export_slam_graph slam_toolbox rclcpp)
target_link_libraries(export_slam_graph Boost::serialization TBB::tbb ${KARTO_LIBRARY})
install(TARGETS export_slam_graph DESTINATION lib/${PROJECT_NAME})
ament_package()
